# Ranking Shorts Renderer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Render a validated ranking manifest and local assets into a 45–60 second 9:16 draft video with narration, captions, BGM, and reusable motion styles.

**Architecture:** A Python package under the versioned `gohome-study-hub` repository owns deterministic text and motion rendering. It accepts videos produced by Seedance or other sources, so Japanese text and ranking numbers are never generated by the video model; a thin documented entry point makes the package available to the existing fal-seedance workflow.

**Tech Stack:** Python 3.12+, Pillow, stdlib `unittest`, ffmpeg/ffprobe, Gemini TTS HTTP API.

## Global Constraints

- Input manifests must pass the schema from the data plan and contain exactly ranks 1, 2, and 3.
- Standard canvas is 1080x1920 at 30fps; 720x1280 is permitted only through an explicit CLI option.
- Text is rendered with a bundled Japanese font; Seedance never renders Japanese text or ranking numbers.
- Special transitions are limited to two per video.
- Initial motions are `cutout-zoom`, `split-reveal`, and `letter-scatter`; placements are `hook`, `chapter`, and `none`.
- Narration uses kanji/kana text, while rendered captions match the approved script exactly.
- Final output is H.264/AAC and must fully decode.

```python
from dataclasses import dataclass
import json
from pathlib import Path

MOTIONS = {"cutout-zoom", "split-reveal", "letter-scatter"}
PLACEMENTS = {"hook", "chapter", "none"}

@dataclass(frozen=True)
class RankingItem:
    rank: int
    content_id: str
    title: str
    url: str
    metric_value: int
    secondary_metric_value: int

@dataclass(frozen=True)
class RankingManifest:
    channel: str
    month: str
    ranking_metric: str
    ranking_label: str
    items: tuple[RankingItem, ...]

    @classmethod
    def from_path(cls, path: Path):
        raw = json.loads(path.read_text(encoding="utf-8"))
        items = tuple(RankingItem(int(x["rank"]), str(x["contentId"]), str(x["title"]), str(x["url"]), int(x["metricValue"]), int(x.get("secondaryMetricValue", 0))) for x in raw.get("items", []))
        if [x.rank for x in items] != [1, 2, 3]:
            raise ValueError("manifest must contain exactly ranks 1, 2, 3")
        return cls(str(raw["channel"]), str(raw["period"]["month"]), str(raw["rankingMetric"]), str(raw["rankingLabel"]), items)

@dataclass(frozen=True)
class RenderConfig:
    width: int = 1080
    height: int = 1920
    fps: int = 30
    placement: str = "hook"
    motion: str = "cutout-zoom"

    def __post_init__(self):
        if self.motion not in MOTIONS: raise ValueError(f"unknown motion: {self.motion}")
        if self.placement not in PLACEMENTS: raise ValueError(f"unknown placement: {self.placement}")
        if (self.width, self.height) not in {(1080, 1920), (720, 1280)}: raise ValueError("unsupported canvas")
        if self.fps != 30: raise ValueError("fps must be 30")
```

```python
import math

def clamp01(value): return max(0.0, min(1.0, float(value)))
def ease_in(value):
    t = clamp01(value)
    return t * t * t
def ease_in_out(value):
    t = clamp01(value)
    return 0.5 - math.cos(math.pi * t) / 2
```

```python
from .easing import ease_in, ease_in_out

def motion_state(name, progress, glyph_count):
    t = max(0.0, min(1.0, progress))
    if name == "cutout-zoom": return {"scale": 1 + 8 * ease_in(t), "opacity": 1}
    if name == "split-reveal":
        v = ease_in_out(t)
        return {"top_y": -v, "bottom_y": v, "gap": v}
    if name == "letter-scatter":
        v = ease_in_out(t)
        glyphs = tuple({"x": ((i % 2) * 2 - 1) * v, "y": (i - (glyph_count - 1) / 2) * .35 * v, "rotation": ((i * 37) % 90 - 45) * v, "scale": 1 - .25 * v} for i in range(glyph_count))
        return {"glyphs": glyphs, "opacity": 1 - v}
    raise ValueError(f"unknown motion: {name}")
```

```python
from PIL import Image, ImageDraw, ImageFilter, ImageFont
from pathlib import Path

FONT = Path(__file__).parents[3] / "public/fonts/IPAexGothic.ttf"

def fit_asset(image, canvas):
    background = image.copy().resize(canvas, Image.Resampling.LANCZOS).filter(ImageFilter.GaussianBlur(28))
    scale = min(canvas[0] / image.width, canvas[1] / image.height)
    foreground = image.resize((round(image.width * scale), round(image.height * scale)), Image.Resampling.LANCZOS)
    background.paste(foreground, ((canvas[0] - foreground.width) // 2, (canvas[1] - foreground.height) // 2))
    return background

def semantic_lines(text, limit):
    if len(text) <= limit: return [text]
    result = []
    remaining = text
    while len(remaining) > limit:
        candidates = [remaining.rfind(mark, 0, limit + 1) + 1 for mark in "、。！？・ " ]
        cut = max(candidates) if max(candidates) > 0 else limit
        result.append(remaining[:cut])
        remaining = remaining[cut:]
    if remaining: result.append(remaining)
    assert "".join(result) == text
    return result
```
